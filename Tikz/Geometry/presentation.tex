            % Define coordinates

            \coordinate (B) at (\a,0,0);
            \coordinate (C) at (\a,\a,0);

            \draw[->] (0,0,0) -- (0,\a+1,0) node[anchor=south west]{$y$};
            \draw[->] (0,0,0) -- (0,0,\hs+1) node[anchor=south]{$z$};

            % Draw base
            \draw[dashed] (A) -- (D);
            \draw[dashed] (D) -- (C);
            \draw (A) -- (B);
            \draw (B) -- (C);

            % Draw height
            \draw[dashed] (S) -- (I);

            % Draw edges from S
            \draw (S) -- (A);
            \draw (S) -- (B);
            \draw (S) -- (C);
            \draw (S) -- (D);

            % Draw projection of SC onto base
            \draw[dashed] (I) -- (C);
            \tdplotsetthetaplanecoords{0} % Set plane for angle
            \tdplotdrawarc[tdplot_rotated_coords, ->]{(I)}{0.5}{0}{atan2(\hs,veclen(\a/2,\a))}{anchor=north east, color=blue}{$45^\circ$}

            % Labels

            \node at (B) [below right] {$B$};
            \node at (C) [below right] {$C$};
            \node at (D) [above left] {$D$};
            \node at (S) [above] {$S$};
            \node at (I) [below] {$I$};
            \node at ($(\a/2,0,0)!0.5!(\a,0,0)$) [below] {$a$};
            \node at ($(0,\a/2,0)!0.5!(0,\a,0)$) [left] {$a$};

            % Add dots to vertices
            \foreach \p in {A,B,C,D,S,I} {
                \filldraw[black] (\p) circle (1.5pt);
            }
